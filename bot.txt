def load_data():
    if not os.path.exists(DB_FILE):
        return {
            "users": {},
            "requests": {}
        }

    with open(DB_FILE, "r") as f:
        return json.load(f)

def save_data(data):
    with open(DB_FILE, "w") as f:
        json.dump(data, f, indent=4)

def get_user(uid):
    data = load_data()

    uid = str(uid)

    if uid not in data["users"]:
        data["users"][uid] = {
            "balance": 0
        }
        save_data(data)
    return data["users"][uid]

@bot.message_handler(func=lambda m: True)
def all_messages(message):

    uid = message.from_user.id

    if uid not in user_state:
        return

    state = user_state[uid]

    # TRANSFER USER ID

    if state["step"] == "transfer_user":

        if not message.text.isdigit():
            bot.send_message(
                message.chat.id,
                "❌ Valid User ID দিন"
            )
            return

        user_state[uid] = {
            "step": "transfer_amount",
            "receiver": int(message.text)
        }

        bot.send_message(
            message.chat.id,
            "🪙 কত Coin Transfer করবেন?"
        )

        return

    # TRANSFER AMOUNT

    if state["step"] == "transfer_amount":

        if not message.text.isdigit():
            bot.send_message(
                message.chat.id,
                "❌ শুধু সংখ্যা লিখুন"
            )
            return

        amount = int(message.text)

        sender = get_user(uid)

        if sender["balance"] < amount:
            bot.send_message(
                message.chat.id,
                "❌ Balance যথেষ্ট নয়"
            )

            del user_state[uid]
            return

        user_state[uid] = {
            "step": "transfer_ref",
            "receiver": state["receiver"],
            "amount": amount
        }

        bot.send_message(
            message.chat.id,
            "📝 Ref লিখুন\nনা চাইলে - লিখুন"
        )

        return

    # TRANSFER REF

    if state["step"] == "transfer_ref":

        receiver = state["receiver"]
        amount = state["amount"]
        ref = message.text

        data = load_data()

        sender_id = str(uid)
        receiver_id = str(receiver)

        if receiver_id not in data["users"]:
            data["users"][receiver_id] = {
                "balance": 0
            }

        data["users"][sender_id]["balance"] -= amount
        data["users"][receiver_id]["balance"] += amount

        save_data(data)

        try:
            bot.send_message(
                receiver,
                f"💸 Balance Received\n\n"
                f"👤 From: {uid}\n"
                f"🪙 Coin: {amount}\n"
                f"📝 Ref: {ref}"
            )
        except:
            pass

        bot.send_message(
            message.chat.id,
            f"✅ Transfer Successful\n\n"
            f"👤 To: {receiver}\n"
            f"🪙 Coin: {amount}\n"
            f"📝 Ref: {ref}"
        )

        del user_state[uid]

        return
# BUY COIN AMOUNT

    if state["step"] == "buy_amount":

        if not message.text.isdigit():
            bot.send_message(
                message.chat.id,
                "❌ শুধু সংখ্যা লিখুন"

@bot.callback_query_handler(func=lambda call: True)
def callback_handler(call):

    data = call.data

    # ==========================
    # JOIN CHECK
    # ==========================
    if data == "check_join":

        user_id = call.from_user.id

        try:
            admin = bot.get_chat_member(ADMIN_CHANNEL, user_id)
            details = bot.get_chat_member(DETAILS_CHANNEL, user_id)

            ok = ["member", "administrator", "creator"]

            if admin.status in ok and details.status in ok:

                get_user(user_id)

                bot.delete_message(
                    call.message.chat.id,
                    call.message.message_id
                )

                bot.send_message(
                    call.message.chat.id,
                    "✅ Verification Successful!",
                    reply_markup=main_menu()
                )

            else:

                bot.answer_callback_query(
                    call.id,
                    "❌ আগে সব চ্যানেলে Join করুন!",
                    show_alert=True
                )

        except Exception as e:

            bot.answer_callback_query(
                call.id,
                f"Error: {e}",
                show_alert=True
            )

    # ==========================
    # APPROVE
    # ==========================
    elif data.startswith("approve:"):

        uid = data.split(":")[1]

        if uid not in buy_requests:
            return

        req = buy_requests[uid]

        db = load_data()

        uid_str = str(req["user_id"])

        if uid_str not in db["users"]:
            db["users"][uid_str] = {
                "balance": 0
            }

        db["users"][uid_str]["balance"] += req["coin"]

        save_data(db)

        bot.send_message(
            req["user_id"],
            f"✅ Order Approved\n\n🪙 Added: {req['coin']} Coin"
        )

        bot.edit_message_text(
            f"✅ Approved\nUser: {uid}",
            call.message.chat.id,
            call.message.message_id
        )

        del buy_requests[uid]

    # ==========================
    # REJECT
    # ==========================
    elif data.startswith("reject:"):

        uid = data.split(":")[1]

        if uid not in buy_requests:
            return

        req = buy_requests[uid]

        bot.send_message(
            req["user_id"],
            "❌ Order Rejected"
        )

        bot.edit_message_text(
            f"❌ Rejected\nUser: {uid}",
            call.message.chat.id,
            call.message.message_id
        )

        del buy_requests[uid]

    elif data.startswith("reject:"):

        uid = data.split(":")[1]

        if uid not in buy_requests:
            return

        req = buy_requests[uid]

        bot.send_message(
            req["user_id"],
            "❌ Order Rejected"
        )

        bot.edit_message_text(
            f"❌ Rejected\nUser: {uid}",
            call.message.chat.id,
            call.message.message_id
        )

        del buy_requests[uid]
